cmake_minimum_required(VERSION 3.16)
project(limo_autorace_sim)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
ament_python_install_package(${PROJECT_NAME})
set(WEGO_DESCRIPTION "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/limo_pro/limo_ros2/limo_description")
if(NOT EXISTS "${WEGO_DESCRIPTION}/meshes/limo_base.dae")
  message(FATAL_ERROR "Run tools/fetch_vehicle.sh first. Upstream meshes are a separate, version-pinned dependency.")
endif()
install(DIRECTORY config launch urdf worlds models rviz DESTINATION share/${PROJECT_NAME})
install(FILES "${WEGO_DESCRIPTION}/meshes/limo_base.dae" "${WEGO_DESCRIPTION}/meshes/limo_wheel.dae" DESTINATION share/${PROJECT_NAME}/meshes)
install(PROGRAMS scripts/command_guard scripts/sensor_delay scripts/teleop scripts/smoke_test DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_pytest REQUIRED)
 ament_add_pytest_test(core test/test_core.py)
endif()
ament_package()
