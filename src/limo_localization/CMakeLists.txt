cmake_minimum_required(VERSION 3.16)
project(limo_localization)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs calib3d)
find_package(Eigen3 REQUIRED)
foreach(dep rclcpp sensor_msgs nav_msgs geometry_msgs diagnostic_msgs std_msgs tf2_ros cv_bridge)
  find_package(${dep} REQUIRED)
endforeach()
add_library(localization_core src/core.cpp)
target_include_directories(localization_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(localization_core PUBLIC ${OpenCV_LIBS} Eigen3::Eigen)
add_executable(road_marking_localizer src/localizer_node.cpp)
target_link_libraries(road_marking_localizer localization_core)
ament_target_dependencies(road_marking_localizer rclcpp sensor_msgs nav_msgs geometry_msgs diagnostic_msgs std_msgs tf2_ros cv_bridge)
install(TARGETS road_marking_localizer DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(localization_core_test test/test_core.cpp)
  target_link_libraries(localization_core_test localization_core)
  add_test(NAME localization_core COMMAND localization_core_test)
endif()
ament_package()
